## ?? **IMPLEMENTATION WITH CLINICAL ROLES**

### **Backend .NET 9 Services**

```csharp
// Claim Review Service
[Authorize(Roles = "TECHNICAL_REVIEWER,MEDICAL_CODER,CLINICAL_REVIEWER_NURSE,MEDICAL_DIRECTOR")]
public class ClaimReviewService
{
    // Technical Review methods
    [Authorize(Roles = "TECHNICAL_REVIEWER")]
    public async Task<TechnicalReviewResult> PerformTechnicalReview(int claimId) { }
    
    // Medical Coding methods
    [Authorize(Roles = "MEDICAL_CODER")]
    public async Task<CodingReviewResult> PerformCodingReview(int claimId) { }
    
    // Clinical Review methods
    [Authorize(Roles = "CLINICAL_REVIEWER_NURSE")]
    public async Task<ClinicalReviewResult> PerformClinicalReview(int claimId) { }
    
    // Medical Director Override
    [Authorize(Roles = "MEDICAL_DIRECTOR")]
    public async Task<Result> OverrideDecision(int claimId, string reason) { }
    
    // QA Audit methods
    [Authorize(Roles = "QA_TECHNICIAN")]
    public async Task<QAAuditResult> AuditReview(int claimId) { }
}

// Claim Review Process Manager
public class ClaimReviewProcessManager
{
    public async Task<ClaimReviewResult> ProcessClaimThroughReview(int claimId)
    {
        // Step 1: Technical Review (TECHNICAL_REVIEWER)
 var technicalResult = await _reviewService.PerformTechnicalReview(claimId);
        if (!technicalResult.Passed)
         return ClaimReviewResult.Rejected(technicalResult.Reason);
        
   // Step 2: Parallel - Coding & Clinical Reviews
        var codingTask = _reviewService.PerformCodingReview(claimId);    // MEDICAL_CODER
      var clinicalTask = _reviewService.PerformClinicalReview(claimId); // CLINICAL_REVIEWER_NURSE
        
        await Task.WhenAll(codingTask, clinicalTask);
        
     // Step 3: Compile Results & Determine Final Status
        var clinicalResult = await clinicalTask;
      
        if (clinicalResult.RequiresDirectorApproval)
 {
      // Escalate to Medical Director
            return ClaimReviewResult.PendingDirector();
        }
     
        // Step 4: QA Audit (QA_TECHNICIAN)
   var qaResult = await _reviewService.AuditReview(claimId);
        
    return ClaimReviewResult.Success(clinicalResult.Status);
    }
}
```
